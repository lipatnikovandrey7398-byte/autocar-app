    function openChatWindow(chatId, carId, sellerPhone, carBrand, carModel, sellerName){
        const mainApp = document.getElementById('mainApp');
        mainApp.innerHTML = `
        <div class="chat-window">
            <div class="detail-header">
                <h2><i class="fas fa-comments"></i> Чат: ${carBrand} ${carModel}</h2>
                <button id="backToChatsList" class="nav-btn" style="background:white;color:var(--primary-blue);"><i class="fas fa-arrow-left"></i> К списку чатов</button>
            </div>
            <div class="chat-messages-area" id="chatMessagesArea"></div>
            <div class="chat-input-area">
                <input type="text" id="chatMessageInput" placeholder="Введите сообщение...">
                <button id="sendMessageBtn" class="btn-primary" style="width:auto;padding:12px 24px;">Отправить</button>
            </div>
        </div>`;
        function renderMessages(){
            const container = document.getElementById('chatMessagesArea');
            if(!container) return;
            container.innerHTML = '';
            chatMessages[chatId].forEach(msg => {
                const isMy = msg.sender === currentUser.phone;
                const div = document.createElement('div');
                div.className = `chat-message ${isMy ? 'my' : 'other'}`;
                div.innerHTML = `<div class="message-bubble"><div style="font-size:0.7rem;font-weight:600;margin-bottom:4px;">${msg.senderName}</div>${msg.text}<div style="font-size:0.7rem;margin-top:4px;">${new Date(msg.time).toLocaleTimeString().slice(0,5)}</div></div>`;
                container.appendChild(div);
            });
            container.scrollTop = container.scrollHeight;
        }
        renderMessages();
        const sendBtn = document.getElementById('sendMessageBtn');
        const input = document.getElementById('chatMessageInput');
        const sendHandler = () => {
            const text = input.value.trim();
            if(!text) return;
            chatMessages[chatId].push({ sender: currentUser.phone, senderName: currentUser.name, text, time: Date.now() });
            saveChats();
            renderMessages();
            input.value = '';
            updateChatsBadge();
        };
        sendBtn.onclick = sendHandler;
        input.onkeypress = (e) => { if(e.key === 'Enter') sendHandler(); };
        document.getElementById('backToChatsList').onclick = () => showChatsPage();
    }
